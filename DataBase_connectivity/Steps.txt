step2:
	open visual studio and install the package of sql conn
	Tools->NuGet Package Manger-> Package Manger Console
step 3:
	Install package 
	Install-Package MySql.Data
step 4:
	open Web.config
	Delete all the data writen in the file and write the code
	"
		<configuration>
		<connectionStrings>
			<add name="MySqlConn"
				 connectionString="server=localhost;user id=root;password=;database=d35"
				 providerName="MySql.Data.MySqlClient"/>
		</connectionStrings>
			<system.web>
				<compilation debug="true" targetFramework="4.8"></compilation>
			</system.web>
		</configuration>
	"

step 5:
